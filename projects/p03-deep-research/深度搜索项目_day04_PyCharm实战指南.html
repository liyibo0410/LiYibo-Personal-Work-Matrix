一深度搜索项目介绍

本项目是 DeepAgents 框架的一个典型最佳实践，目标是构建一个「深度搜索研究员」：模拟人类高级研究员的思维，把「搜索 → 阅读 → 反思 → 再搜索」的多轮迭代过程自动化，实现广覆盖、高精准、强可靠的复杂信息处理与文档生成。

1.1 项目目标

利用 DeepAgents 构建多路组合智能体系统，以「主智能体统筹 + 多专家子智能体并行协作」为核心架构，突破传统 RAG 单次检索的局限，通过多轮迭代深度挖掘海量信息背后的隐藏逻辑，最终自动产出高质量的 Markdown / PDF 分析文档。

1.2 Agent Design —— 1 主 + N 专

系统采用「1 主 + N 专」多路组合模式：主智能体是「项目经理」，不亲自干活，只负责理解需求 → 拆解任务 → 调度资源 → 交付结果；三类专家子智能体各司其职、并行工作。

2.4.2 API 通信层（api/ 包）

通信层解决两个核心问题：① 多用户/多会话隔离——每个请求带一个 thread_id，ContextVar 保证并发不串数据；② 实时进度推送——WebSocket 管理器把 Agent 的思考过程推给前端。

api/context.py — 会话上下文隔离核心：ContextVar
"""会话上下文管理：用 ContextVar 做线程/协程级别的变量隔离"""
from contextvars import ContextVar    # 上下文变量：每个异步任务有自己的值
from typing import Optional

# 当前会话的 thread_id（哪个用户在提问）
_thread_id_var: ContextVar[Optional[str]] = ContextVar("thread_id", default=None)
# 当前会话的输出目录（output/session_xx）
_session_dir_var: ContextVar[Optional[str]] = ContextVar("session_dir", default=None)

def set_thread_context(thread_id: str):
    """保存 thread_id，返回 token 用于后续恢复"""
    return _thread_id_var.set(thread_id)      # set 返回一个 token

def get_thread_context() -> Optional[str]:
    """读取当前会话的 thread_id"""
    return _thread_id_var.get()

def set_session_context(session_dir: str):
    """保存当前会话的输出目录"""
    return _session_dir_var.set(session_dir)

def get_session_context() -> Optional[str]:
    """读取当前会话的输出目录（工具保存文件时要用）"""
    return _session_dir_var.get()

def reset_session_context(session_token, thread_token):
    """任务结束：用 token 恢复旧值，防止变量泄漏到下一个请求"""
    _session_dir_var.reset(session_token)     # 撤销本次 set
    _thread_id_var.reset(thread_token)

api/monitor.py — WebSocket 管理器与进度上报核心：ConnectionManager
"""实时监控：管理 WebSocket 连接，把智能体进度推送给前端"""
import asyncio
import json
from typing import Dict
from fastapi import WebSocket

class ConnectionManager:
    """WebSocket 连接管理器：按 thread_id 分组维护长连接"""

    def __init__(self):
        # thread_id -> 该会话的所有 WebSocket 连接
        self.active_connections: Dict[str, list[WebSocket]] = {}
        # 后端运行的事件循环（startup 事件里绑定，线程安全投递用）
        self.loop = None

    def set_loop(self, loop):
        """服务启动时绑定主事件循环"""
        self.loop = loop

    async def connect(self, websocket: WebSocket, thread_id: str):
        """接受连接并注册到对应会话分组"""
        await websocket.accept()                 # 完成 WS 握手
        if thread_id not in self.active_connections:
            self.active_connections[thread_id] = []
        self.active_connections[thread_id].append(websocket)

    def disconnect(self, websocket: WebSocket, thread_id: str):
        """连接断开时从分组中移除"""
        if thread_id in self.active_connections:
            self.active_connections[thread_id].remove(websocket)

    async def _send_personal_message(self, message: dict, websocket: WebSocket):
        """向单个连接发送 JSON"""
        try:
            await websocket.send_json(message)
        except Exception:
            pass                                # 客户端可能已断开，静默忽略

    def _emit(self, event: str, message: str, thread_id: str | None = None):
        """向某会话的所有连接广播事件（核心方法）"""
        payload = {"event": event, "message": message}
        # 没绑定 loop（如本地脚本测试）→ 打印到控制台即可
        if self.loop is None:
            print(f"[monitor] {event} | {message}")
            return
        for tid, conns in self.active_connections.items():
            if thread_id and tid != thread_id:
                continue                        # 只发给指定会话
            for ws in conns:
                # 协程投递到事件循环线程执行（线程安全）
                asyncio.run_coroutine_threadsafe(
                    self._send_personal_message(payload, ws), self.loop)

    # ---- 以下是对外便捷方法：事件类型语义化 ----
    def report_assistant(self, subagent_type: str, description: str):
        """上报：主智能体准备调用某子智能体"""
        self._emit("assistant", f"调用子智能体【{subagent_type}】: {description}")

    def report_tool(self, tool_name: str, args=None):
        """上报：智能体准备调用某工具"""
        self._emit("tool", f"调用工具【{tool_name}】: {args}")

    def report_task_result(self, content: str):
        """上报：智能体返回最终结果"""
        self._emit("result", content)

    def report_session_dir(self, session_dir: str):
        """上报：当前会话的输出目录（前端用于展示文件列表）"""
        self._emit("session_dir", session_dir)

# 全局单例：server.py 与各工具都 import 这一个实例
manager = ConnectionManager()
# 兼容旧命名：很多工具文件里写的是 monitor
monitor = manager

    root.setLevel(level)
    root.addHandler(console)

"""路径工具：把相对路径安全解析到会话目录内，防止路径穿越"""
from pathlib import Path
import os

def resolve_path(filename: str, session_dir: str) -> str:
    """
    将 filename 解析为 session_dir 下的绝对路径。
    核心：无论传绝对路径还是相对路径，最终都收拢到会话目录里，
    杜绝 `../../etc/passwd` 这类越权访问。
    """
    if not session_dir:
        session_dir = "."                        # 兜底：未配置就用当前目录
    session_path = Path(session_dir).resolve()   # 会话目录的绝对路径
    input_path = Path(filename)                  # 目标文件名/路径

    # ① 绝对路径：直接拼到会话目录下（丢弃盘符/根前缀）
    if input_path.is_absolute():
        full_path = session_path / input_path.name
    # ② 相对路径：文件名可能含子目录（如 sub_dir/a.md）
    else:
        full_path = session_path / input_path

    # ③ 安全校验：解析后必须仍在会话目录内
    full_path = full_path.resolve()
    if not str(full_path).startswith(str(session_path)):
        raise ValueError(f"路径越界: {full_path}")
    return str(full_path)

from utils.logger_utils import logger          # 日志
import os
from dotenv import load_dotenv                  # .env 加载
from typing import Literal                      # 限定枚举类型
from langchain_core.tools import tool           # LangChain 工具装饰器
from tavily import TavilyClient                 # Tavily 官方客户端
from api.monitor import monitor                 # 进度上报

load_dotenv()                                   # 加载 .env
_tavily_client = TavilyClient(api_key=os.getenv("TAVILY_API_KEY"))  # 全局客户端

@tool
def internet_search(
        query: str,                                              # ① 查询内容
        topic: Literal["general", "news", "finance"] = "general",# ② 类别（枚举）
        max_results: int = 5,                                    # ③ 返回条数
        include_raw_content: bool = False                        # ④ 是否含正文
):
    """
    进行网络搜索工具,可以查询数据库或者rag助手中不包含的数据
    :param query: 查询内容
    :param topic: 查询类别
    :param max_results: 查询返回的条数
    :param include_raw_content: 是否精简返回
    :return: 查询的返回结果
    """
    # 1. 日志记录（调试排查用）
    logger.debug(f"开始进行internet_search工具的调用!查询内容:{query} , 查询数量:{max_results}")
    # 2. 向前端推送「正在调用工具」事件
    monitor.report_tool(tool_name="网络搜索工具", args={"query": query, "topic": topic,
                                                       "max_results": max_results,
                                                       "include_raw_content": include_raw_content})
    # 3. 真正调用 Tavily 搜索
    return _tavily_client.search(
        query=query,
        topic=topic,
        max_results=max_results,
        include_raw_content=include_raw_content
    )

tools/mysql_tools.py — 数据库三件套
import os
from dotenv import load_dotenv
from api.monitor import monitor
from mysql.connector import connect, Error      # MySQL 官方驱动
from typing import Annotated, List
from langchain_core.tools import tool

load_dotenv()

# ---- 从环境变量组装连接配置 ----
def get_db_config():
    """Get database configuration from environment variables."""
    config = {
        "host": os.getenv("MYSQL_HOST", "localhost"),
        "port": int(os.getenv("MYSQL_PORT", "3306")),
        "user": os.getenv("MYSQL_USER"),
        "password": os.getenv("MYSQL_PASSWORD"),
        "database": os.getenv("MYSQL_DATABASE"),
        "charset": os.getenv("MYSQL_CHARSET", "utf8mb4"),
        "collation": os.getenv("MYSQL_COLLATION", "utf8mb4_unicode_ci"),
        "autocommit": True,                      # 自动提交（只读场景安全）
        "sql_mode": os.getenv("MYSQL_SQL_MODE", "TRADITIONAL")
    }
    # 移除 None 值（没配置的键不传给驱动，避免报错）
    config = {k: v for k, v in config.items() if v is not None}
    # 校验核心配置是否齐全
    required_keys = ["user", "password", "database"]
    missing_keys = [k for k in required_keys if k not in config]
    if missing_keys:
        raise ValueError(f"缺失数据库核心配置：{', '.join(missing_keys)}")
    return config

# ============ 1. 列出所有表 ============
@tool
def list_sql_tables() -> str:
    """查询当前库中有哪些可用表! 对应的sql: show tables;"""
    monitor.report_tool(tool_name="搜索表名工具")
    try:
        with connect(**get_db_config()) as conn:      # with：用完自动关连接
            with conn.cursor() as cursor:             # 游标：执行 SQL 的入口
                sql = "show tables;"
                cursor.execute(sql)                   # 执行查询
                select_result = cursor.fetchall()     # 取回全部行 [(a,), (b,)]
                if not select_result or len(select_result) == 0:
                    return "没有可用表"
                table_name_list = [item[0] for item in select_result]  # 取每行第0列
                return f"可用表:{','.join(table_name_list)}"
    except Error as e:
        return "数据库连接失败: " + str(e)

# ============ 2. 预览表数据（前100行） ============
@tool
def get_table_data(table_name: str) -> str:
    """查询指定表名的数据，返回表格格式文本，让助手理解表结构"""
    monitor.report_tool(tool_name="搜索指定表名的数据工具", args={"table_name": table_name})
    try:
        with connect(**get_db_config()) as conn:
            with conn.cursor() as cursor:
                sql = f"select * from {table_name} limit 100;"
                cursor.execute(sql)
                description = cursor.description       # 列元数据（表头）
                if not description:
                    return f"当前表{table_name}没有数据"
                table_header = [c[0] for c in description]          # 列名列表
                result = cursor.fetchall()                           # 行元组列表
                data_list = [",".join(map(str, row)) for row in result]  # 行→"a,b,c"
                return ",".join(table_header) + "\n" + "\n".join(data_list)
    except Error as e:
        return "数据库连接失败: " + str(e)

# ============ 3. 执行自定义 SQL ============
@tool
def execute_sql_query(sql: str) -> str:
    """执行自定义SQL查询（复杂筛选/联接/聚合）"""
    monitor.report_tool(tool_name="执行自定义SQL查询工具", args={"sql": sql})
    try:
        with connect(**get_db_config()) as conn:
            with conn.cursor() as cursor:
                cursor.execute(sql)
                description = cursor.description
                if not description:
                    return f"当前语句{sql}没有数据"
                table_header = [c[0] for c in description]
                result = cursor.fetchall()
                data_list = [",".join(map(str, row)) for row in result]
                return ",".join(table_header) + "\n" + "\n".join(data_list)
    except Error as e:
        return "数据库连接失败: " + str(e)

if __name__ == '__main__':
    # 本地自测：直接验证 SQL 工具
    print(execute_sql_query.invoke({"sql": "select * from drugs limit 100"}))

with connect(...) as conn上下文管理器自动关闭连接，防资源泄漏——每个工具都复用这个模式
cursor.description返回结果集的列元数据，取 [0] 就是列名——用它动态生成「表头行」

        chat.delete_sessions([session.id])
        return final_result
    except Exception as e:
        return f"向{chat_name}提问失败：{str(e)}"

3.4.1 MainAgent 提示词（prompts.yml 细化版）

主智能体提示词有几个红线规则必须遵守：

🧭
信息获取规则

外部/背景知识→网络助手；企业内部非流通知识→RAGFlow；企业商品数据→数据库助手；边界不明→三个全用。

📁
工作目录强制

⏳

必须先取信息再生成文件；严禁一步同时搜索+生成；严禁占位符生成；文档须含 todo-list 且不少于 1000 字。

① 读取上传文件工具（tools/upload_file_read_tool.py）——支持 .md/.txt/.docx/.pdf/.xlsx/.xls 六类文件，Excel 还会附带统计信息：

tools/upload_file_read_tool.py — 读文件
from pathlib import Path
from typing import Annotated
from langchain_core.tools import tool
from api.monitor import monitor
from api.context import get_session_context
from utils.path_utils import resolve_path
import docx                      # Word 解析
import pypdf                     # PDF 解析
import pandas as pd              # Excel 解析

@tool
def read_file_content(
        filename: Annotated[str, "要读取的文件名或路径（支持 .md, .docx, .pdf, .xlsx, .xls）"],
        instruction: Annotated[str, "对提取内容的具体指令（例如：'提取摘要', '统计数据'）"] = "提取全部内容"
) -> str:
    """读取指定文件的内容。支持 Markdown(.md)、Word(.docx)、PDF(.pdf) 和 Excel(.xlsx/.xls)。"""
    monitor.report_tool("文件内容读取工具", {"filename": filename, "instruction": instruction})

    # ===== 1. 路径解析（收拢到当前会话目录，防穿越） =====
    session_dir = get_session_context()                       # 从 ContextVar 取会话目录
    file_path = Path(resolve_path(filename, session_dir))     # 安全解析
    if not file_path.exists():
        return f"错误：文件 '{filename}' 不存在 (解析路径: {file_path})。"
    ext = file_path.suffix.lower()                            # 后缀转小写

    try:
        # ===== 2. 按类型读取 =====
        if ext in ['.md', '.txt']:
            return file_path.read_text(encoding='utf-8')      # 纯文本直接读

        elif ext == '.docx':
            doc = docx.Document(str(file_path))               # Word：取所有段落
            return '\n'.join([p.text for p in doc.paragraphs])

        elif ext == '.pdf':
            reader = pypdf.PdfReader(str(file_path))          # PDF：逐页提取文本
            return "\n".join([page.extract_text() or "" for page in reader.pages])

        elif ext in ['.xlsx', '.xls']:
            df = pd.read_excel(str(file_path))                # Excel：转 DataFrame
            result = [
                f"文件: {filename}",
                f"行数: {len(df)}, 列数: {len(df.columns)}",
                f"列名: {', '.join(df.columns.astype(str))}",
                "\n[前5行数据预览]:", df.head().to_string(index=False),
                "\n[统计描述]:", df.describe().to_string()
            ]
            return "\n".join(result)                          # 结构+预览+统计

        else:
            return file_path.read_text(encoding='utf-8')      # 兜底按文本读
    except UnicodeDecodeError:
        return f"错误：不支持的文件格式 '{ext}'，且无法作为文本读取。"
    except Exception as e:
        return f"读取文件出错: {str(e)}"

utils/converter_utils.py — 核心转换逻辑
from utils.logger_utils import logger
from pathlib import Path
import markdown                      # Markdown → HTML
import win32com.client               # 调用 Word 的 COM 接口（Windows）
import pythoncom                     # COM 初始化/释放

def convert_md_to_pdf_via_word(md_abs_path: Path, pdf_abs_path: Path) -> str:
    """使用 Microsoft Word COM 接口将 Markdown 转换为 PDF。"""
    temp_html_path = md_abs_path.with_suffix('.temp.html')   # 中间产物
    word_app = None

    try:
        # ===== 1. MD → HTML（带表格/代码块扩展） =====
        with open(md_abs_path, 'r', encoding='utf-8') as f:
            md_content = f.read()
        html_body = markdown.markdown(md_content, extensions=['tables', 'fenced_code'])
        html_content = f"""
        
        {html_body}
        """
        with open(temp_html_path, 'w', encoding='utf-8') as f:
            f.write(html_content)

        # ===== 2. 用 Word COM 打开 HTML，另存为 PDF =====
        pythoncom.CoInitialize()                         # 初始化 COM 线程
        word_app = win32com.client.Dispatch('Word.Application')  # 启动 Word
        word_app.Visible = False                         # 后台运行（不弹窗）
        word_app.DisplayAlerts = False                   # 关掉警告弹窗
        doc = word_app.Documents.Open(str(temp_html_path.resolve()))
        doc.SaveAs(str(pdf_abs_path.resolve()), FileFormat=17)  # 17 = wdFormatPDF
        doc.Close(SaveChanges=0)

        if pdf_abs_path.exists():
            return f"成功转换: {pdf_abs_path} (Word引擎)"
        return f"转换完成但未生成文件: {pdf_abs_path}"

    except ImportError:
        return "缺少依赖库，请安装: pip install pywin32 markdown"
    except Exception as e:
        logger.error(f"Word转换PDF失败: {e}", exc_info=True)
        return f"转换失败: {str(e)}"
    finally:
        # ===== 3. 资源清理：关 Word、删临时 HTML、释放 COM =====
        if word_app:
            try:
                word_app.Quit()
            except Exception as e:
                logger.warning(f"关闭Word应用时出错: {e}")
        if temp_html_path.exists():
            try:
                temp_html_path.unlink()
            except Exception as e:
                logger.warning(f"删除临时HTML文件时出错: {e}")
        try:
            pythoncom.CoUninitialize()
        except Exception as e:
            logger.warning(f"释放COM初始化时出错: {e}")

def _report_agent_process(chunk):
    for key, value in chunk.items():            # chunk: {"model":..., "tools":...}
        if not value or "messages" not in value:
            continue
        message = value["messages"][-1]         # 只看最新一条消息
        if key == "model":
            if message.tool_calls:              # 模型要调用工具/子智能体
                for tool_call in message.tool_calls:
                    if tool_call["name"] == "task":   # task = 调用子智能体
                        monitor.report_assistant(
                            tool_call['args']['subagent_type'],
                            tool_call['args']['description'])
                    else:                       # 其他 = 调用普通工具
                        monitor.report_tool(tool_call['name'], tool_call['args'])
            elif message.content:               # 模型输出最终文本
                monitor.report_task_result(message.content)

# ============ 执行入口 ============
async def run_deep_agent(query: str, thread_id: str):
    # 1. 保存 thread_id 到上下文
    thread_token = set_thread_context(thread_id)
    # 2. 生成会话输出目录 output/session_ 并保存
    project_dir = Path(__file__).parents[1].resolve()       # 项目根
    session_dir = project_dir / "output" / f"session_{thread_id}"
    session_dir.mkdir(parents=True, exist_ok=True)
    session_dir_str = str(session_dir).replace("\\", "/")   # Windows 反斜杠→正斜杠
    session_token = set_session_context(session_dir_str)
    monitor.report_session_dir(session_dir_str)             # 告诉前端输出目录

    # 3. 把 upload/session_ 里的上传文件拷贝到会话目录
    upload_dir = project_dir / "upload" / f"session_{thread_id}"
    uploaded_info = ''
    if upload_dir.exists():
        file_names = [i.name for i in upload_dir.iterdir() if i.is_file()]
        for name in file_names:
            shutil.copy2(upload_dir / name, session_dir / name)
        uploaded_info = (f"\n    [已上传文件] 已加载到工作目录:\n" +
                         "\n".join([f"    - {f}" for f in file_names]) +
                         "\n    请优先使用工具读取并参考这些文件。")

    # 4. 组装「工作环境指令」注入给主智能体
    relative_session_dir_str = str(session_dir.relative_to(project_dir)).replace("\\", "/")
    path_instruction = f"""
            【工作环境指令】
            工作目录: {relative_session_dir_str}
            {uploaded_info}
            规则：
            1. 新生成文件必须保存到工作目录：'{relative_session_dir_str}'
            2. 读取已上传的文件时，请直接将文件名（例如：'开篇.txt'）作为 filename 参数传入
            3. 使用相对路径，禁止使用绝对路径
            4. 若存在上传文件，请先分析内容
            """

    try:
        # 5. 流式执行：遍历每个 chunk 并实时上报给前端
        result = main_agent.astream({
            "messages": [{
                "role": "user",
                "content": f"问题：{query}, 补充描述: {path_instruction}"
            }]
        })
        async for chunk in result:
            _report_agent_process(chunk)
    except Exception as e:
        logger.error(f"智能体执行最终失败：{e}")
        monitor._emit("error", "智能体执行最终失败了")
    finally:
        # 6. 无论成败都重置上下文，防止污染下一个请求
        reset_session_context(session_token, thread_token)

4.5 查询文件列表接口（GET /api/files）

api/server.py — 文件列表
@app.get("/api/files")
async def list_files(path: str):
    """
    文件列表查询接口 (File Explorer)。
    1. 列出指定目录下的所有生成文件。
    2. 提供文件元数据（大小、时间、下载链接）。
    3. 严格的安全检查，防止路径遍历攻击。
    """
    print(f"[DEBUG] 请求文件列表: {path}")

    try:
        abs_path = Path(path).resolve()
        output_abs = _output_dir.resolve()

        # [安全] 检查路径是否越界 (Path Traversal Check)
        if not abs_path.is_relative_to(output_abs):
            print(f"[ERROR] 拒绝访问: {abs_path} 不在 {output_abs} 目录下")
            return {"error": "拒绝访问: 只能访问输出目录下的文件"}
    except Exception as e:
        print(f"[ERROR] 路径解析失败: {e}")
        return {"error": f"路径无效: {e}"}

    # [检查] 目录是否存在
    if not abs_path.exists():
        return {"error": "目录不存在"}

    files = []
    try:
        # [遍历] 递归查找所有文件
        for file_path in abs_path.rglob("*"):
            if file_path.is_file():
                stat = file_path.stat()
                files.append({
                    "name": file_path.name,      # 文件名
                    "type": "file",
                    "path": str(file_path),      # 绝对路径（下载接口用它）
                    "size": stat.st_size,        # 字节数
                    "mtime": stat.st_mtime       # 修改时间戳
                })
    except Exception as e:
        print(f"[ERROR] 遍历文件失败: {e}")
        return {"error": str(e)}

    # [排序] 按修改时间倒序（最新在前）
    files.sort(key=lambda x: x.get("mtime", 0), reverse=True)
    print(f"[DEBUG] 找到 {len(files)} 个文件")
    return {"files": files}
逐行解读
代码作用
rglob("*")递归遍历目录下所有文件（含子目录）
stat()取文件大小与修改时间——前端文件列表卡片的元数据来源
sort reverse=True新生成的文件排最前，用户体验更好

4.6 WebSocket 实时通讯（/ws/{thread_id}）

api/server.py — WebSocket 端点
@app.websocket("/ws/{thread_id}")
async def websocket_endpoint(websocket: WebSocket, thread_id: str):
    """
    WebSocket 实时通讯核心接口。
    1. 建立长连接，实现服务端与前端的双向通信。
    2. 绑定 `thread_id`，实现会话级消息隔离。
    3. 维持心跳 (Keep-Alive)，防止连接超时。
    """
    # 1. [注册] 建立连接并绑定到管理器
    await manager.connect(websocket, thread_id)

    try:
        # 2. [循环] 保持连接活跃
        while True:
            # 3. [监听] 接收前端消息（通常是 ping 心跳）
            data = await websocket.receive_text()

            # 4. [响应] 回复 pong 消息
            await websocket.send_json({
                "type": "pong",
                "message": f"服务端已收到: {data}"
            })
    except WebSocketDisconnect:
        # 5. [清理] 客户端主动断开
        manager.disconnect(websocket, thread_id)
        print(f"[WebSocket] 客户端已断开: {thread_id}")
    except Exception as e:
        # 6. [异常] 发生错误时断开
        print(f"[WebSocket] 连接异常: {e}")
        manager.disconnect(websocket, thread_id)
逐行解读
代码作用
manager.connect把当前 WS 连接按 thread_id 注册进管理器——之后 monitor 广播就能找到它
while True + receive_text长连接循环：前端发 ping 保活，服务端回 pong
WebSocketDisconnect客户端关页/断网时优雅清理，避免僵尸连接占内存
monitor 广播链路Agent 每步 report_* → manager send_to_thread → 前端实时刷新思考过程
